NC='\033[0m' # No Color

if [[ $EUID -eq 0 ]]; then

echo -e "${BLUE}🔍 Checking Python version...${NC}"

required_version="3.11"

if [ "$(printf '%s\n' "$required_version" "$python_version" | sort -V | head -n1)" != "$required_version" ]; then
    echo -e "${RED}❌ Python 3.11+ required. Found: $python_version${NC}"
